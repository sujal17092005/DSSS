#include <stdio.h>
int main() {
    int a[100],n,i,j,t;
    printf("Number of elements: "); scanf("%d",&n);
    if(n<0||n>100) return 0;
    for(i=0;i<n;i++) scanf("%d",&a[i]);
    for(i=0;i<n-1;i++) for(j=0;j<n-i-1;j++)
        if(a[j]>a[j+1]) { t=a[j]; a[j]=a[j+1]; a[j+1]=t; }
    printf("Sorted: ");
    for(i=0;i<n;i++) printf("%d ",a[i]);
    return 0;
}
