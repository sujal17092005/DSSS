struct N{char w[40],meaning[100];struct N*l,*r;};struct N*insert(struct N*t,char w[],char m[]){if(!t){t=malloc(sizeof(*t));strcpy(t->w,w);strcpy(t->meaning,m);t->l=t->r=NULL;}else if(strcmp(w,t->w)<0)t->l=insert(t->l,w,m);else if(strcmp(w,t->w)>0)t->r=insert(t->r,w,m);else strcpy(t->meaning,m);return t;}void print(struct N*t,int rev){if(!t)return;if(rev){print(t->r,rev);printf("%s: %s\n",t->w,t->meaning);print(t->l,rev);}else{print(t->l,rev);printf("%s: %s\n",t->w,t->meaning);print(t->r,rev);}}struct N*search(struct N*t,char w[]){while(t&&strcmp(w,t->w))t=strcmp(w,t->w)<0?t->l:t->r;return t;}int main(){struct N*root=NULL,*p;int ch;char w[40],m[100];do{printf("1 Add 2 Search 3 Ascending 4 Descending 0 Exit: ");scanf("%d",&ch);if(ch==1){scanf(" %39s %99[^\n]",w,m);root=insert(root,w,m);}else if(ch==2){scanf("%39s",w);p=search(root,w);if(p)printf("%s\n",p->meaning);else puts("Not found");}else if(ch==3)print(root,0);else if(ch==4)print(root,1);}while(ch);return 0;}
