#include <stdio.h>
int main() {
    int a[100],n,key,i,low,high,mid,found=-1;
    printf("Number of sorted elements: "); scanf("%d",&n);
    if(n<0||n>100) return 0;
    for(i=0;i<n;i++) scanf("%d",&a[i]);
    printf("Search value: "); scanf("%d",&key);
    low=0; high=n-1;
    while(low<=high) {
        mid=low+(high-low)/2;
        if(a[mid]==key) { found=mid; break; }
        if(a[mid]<key) low=mid+1; else high=mid-1;
    }
    if(found==-1) printf("Not found\n");
    else printf("Found at position %d\n",found+1);
    return 0;
}
