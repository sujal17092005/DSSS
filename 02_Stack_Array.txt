#include <stdio.h>
#define MAX 5
int stack[MAX], top=-1;
void push(int x) { if(top==MAX-1) printf("Overflow\n"); else stack[++top]=x; }
void pop() { if(top==-1) printf("Underflow\n"); else printf("Popped: %d\n",stack[top--]); }
void display() { int i; for(i=top;i>=0;i--) printf("%d ",stack[i]); printf("\n"); }
int main() {
    int ch,x;
    do {
        printf("1 Push  2 Pop  3 Display  0 Exit: ");
        if(scanf("%d",&ch)!=1) break;
        if(ch==1) { printf("Value: "); scanf("%d",&x); push(x); }
        else if(ch==2) pop();
        else if(ch==3) display();
    } while(ch!=0);
    return 0;
}
