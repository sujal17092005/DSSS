#include <stdio.h>
int main() {
    int a[100],n,i,j,min,t;
    printf("Number of elements: "); scanf("%d",&n);
    if(n<0||n>100) return 0;
    for(i=0;i<n;i++) scanf("%d",&a[i]);
    for(i=0;i<n-1;i++) {
        min=i;
        for(j=i+1;j<n;j++) if(a[j]<a[min]) min=j;
        t=a[i]; a[i]=a[min]; a[min]=t;
    }
    printf("Sorted: ");
    for(i=0;i<n;i++) printf("%d ",a[i]);
    return 0;
}
