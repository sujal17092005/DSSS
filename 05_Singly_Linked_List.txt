#include <stdio.h>
#include <stdlib.h>
struct Node { int data; struct Node *next; };
struct Node *head=NULL;
void insert(int x) {
    struct Node *p=malloc(sizeof(struct Node));
    if(!p) return;
    p->data=x; p->next=head; head=p;
}
void deleteFirst() {
    struct Node *p;
    if(!head) { printf("Empty\n"); return; }
    p=head; head=head->next; free(p);
}
void display() {
    struct Node *p=head;
    while(p) { printf("%d -> ",p->data); p=p->next; }
    printf("NULL\n");
}
int main() {
    int ch,x;
    do {
        printf("1 Insert at beginning  2 Delete first  3 Display  0 Exit: ");
        if(scanf("%d",&ch)!=1) break;
        if(ch==1) { printf("Value: "); scanf("%d",&x); insert(x); }
        else if(ch==2) deleteFirst();
        else if(ch==3) display();
    } while(ch!=0);
    while(head) deleteFirst();
    return 0;
}
