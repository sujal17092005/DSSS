#include <stdlib.h>
struct Node{int id;struct Node*next;};struct Node*freeList=NULL,*usedList=NULL;void add(struct Node**h,int id){struct Node*n=malloc(sizeof(struct Node));n->id=id;n->next=*h;*h=n;}int removeid(struct Node**h,int id){struct Node*p=*h,*prev=NULL;while(p&&p->id!=id){prev=p;p=p->next;}if(!p)return 0;if(prev)prev->next=p->next;else *h=p->next;free(p);return 1;}void show(struct Node*h){while(h){printf("%d ",h->id);h=h->next;}puts("");}int main(){int i,ch,id;for(i=1;i<=10;i++)add(&freeList,i);do{printf("1 Allocate 2 Free 3 Show free 4 Show used 0 Exit: ");scanf("%d",&ch);if(ch==1){if(!freeList)puts("No free nodes");else{id=freeList->id;removeid(&freeList,id);add(&usedList,id);printf("Allocated node %d\n",id);}}else if(ch==2){printf("Node ID: ");scanf("%d",&id);if(removeid(&usedList,id))add(&freeList,id);else puts("Not allocated");}else if(ch==3)show(freeList);else if(ch==4)show(usedList);}while(ch);return 0;}
