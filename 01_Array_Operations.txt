#include <stdio.h>
int main() {
    int a[100], n, i, pos, value;
    printf("Enter number of elements: "); scanf("%d", &n);
    if(n < 0 || n > 99) return 0;
    for(i=0;i<n;i++) scanf("%d", &a[i]);
    printf("Enter position (1 to %d) and value to insert: ", n+1);
    scanf("%d%d", &pos, &value);
    if(pos>=1 && pos<=n+1) {
        for(i=n;i>=pos;i--) a[i]=a[i-1];
        a[pos-1]=value; n++;
    }
    printf("Array: ");
    for(i=0;i<n;i++) printf("%d ",a[i]);
    printf("\nEnter position to delete (1 to %d): ",n);
    scanf("%d",&pos);
    if(pos>=1 && pos<=n) {
        for(i=pos-1;i<n-1;i++) a[i]=a[i+1];
        n--;
    }
    printf("Final array: ");
    for(i=0;i<n;i++) printf("%d ",a[i]);
    return 0;
}
